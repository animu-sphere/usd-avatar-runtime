cmake_minimum_required(VERSION 3.22)
project(InstalledUsdBindingConsumer LANGUAGES CXX)
set(components usd)
if(WITH_MOTION)
    list(APPEND components motion)
    if(WITH_VRM)
        list(APPEND components vrm)
    endif()
endif()
find_package(AvatarRuntime 0.1.0 EXACT CONFIG REQUIRED COMPONENTS ${components})
if(TARGET AvatarRuntime::avatarMotionUsdBinding)
    message(FATAL_ERROR "Generic USD binding must not import the source-clip adapter")
endif()
if(NOT TARGET motionUsd::motionUsd)
    message(FATAL_ERROR "Generic USD binding requires owner skeleton reading")
endif()
if(TARGET vrmSchema::vrmSchema OR TARGET AvatarRuntime::avatarVrmUsdBinding)
    message(FATAL_ERROR "Generic USD binding must not import VRM schema discovery")
endif()
add_executable(consumer ../tests.cpp)
target_link_libraries(consumer PRIVATE AvatarRuntime::avatarUsdBinding)
if(WITH_MOTION)
    target_compile_definitions(consumer PRIVATE AR_TEST_MOTION)
    target_link_libraries(consumer PRIVATE AvatarRuntime::avatarMotionAdapter)
    if(WITH_VRM)
        target_compile_definitions(consumer PRIVATE AR_TEST_VRM)
        target_link_libraries(consumer PRIVATE AvatarRuntime::avatarVrmAdapter)
    endif()
endif()
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:consumer>)
endif()
enable_testing()
add_test(NAME installed.usd_binding COMMAND consumer)
