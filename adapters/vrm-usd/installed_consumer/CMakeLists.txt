cmake_minimum_required(VERSION 3.22)
project(InstalledVrmUsdBindingConsumer LANGUAGES CXX)
set(components vrm_usd)
if(WITH_MOTION)
    list(APPEND components motion)
endif()
find_package(AvatarRuntime 0.1.0 EXACT CONFIG REQUIRED COMPONENTS ${components})
if(TARGET motionUsd::motionUsd OR TARGET AvatarRuntime::avatarMotionUsdBinding)
    message(FATAL_ERROR "VRM Humanoid binding must not import source clip reading")
endif()
if(TARGET vrmRig::vrmRig OR TARGET AvatarRuntime::avatarVrmAdapter)
    message(FATAL_ERROR "Schema binding must not import the VRM evaluator")
endif()
if(NOT WITH_MOTION AND (TARGET motionSampling::motionSampling OR TARGET AvatarRuntime::avatarMotionAdapter))
    message(FATAL_ERROR "Schema-only binding must not import motion sampling")
endif()
add_executable(consumer ../tests.cpp)
target_link_libraries(consumer PRIVATE AvatarRuntime::avatarVrmUsdBinding)
if(WITH_MOTION)
    target_compile_definitions(consumer PRIVATE AR_TEST_MOTION)
    target_link_libraries(consumer PRIVATE AvatarRuntime::avatarMotionAdapter)
endif()
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:consumer>)
endif()
enable_testing()
add_test(NAME installed.vrm_usd_binding COMMAND consumer)
