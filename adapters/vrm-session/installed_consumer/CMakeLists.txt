cmake_minimum_required(VERSION 3.22)
project(InstalledVrmSession LANGUAGES CXX)
find_package(AvatarRuntime 0.2.0 EXACT CONFIG REQUIRED)
if(TARGET vrmRig::vrmRig OR TARGET usd OR TARGET motionCore::motionCore)
    message(FATAL_ERROR "Core session lookup imported owner dependencies")
endif()
find_package(AvatarRuntime 0.2.0 EXACT CONFIG REQUIRED COMPONENTS vrm_session)
if(TARGET AvatarRuntime::avatarVrmAdapter OR TARGET AvatarRuntime::avatarStageRunnerSession OR TARGET physicsJolt::physicsJolt)
    message(FATAL_ERROR "VRM session imported unrelated expression/Stage Runner/physics adapters")
endif()
add_executable(consumer ../check.cpp ../tests.cpp)
target_compile_definitions(consumer PRIVATE AR_VRM_SESSION_TEST_ENTRY)
target_link_libraries(consumer PRIVATE AvatarRuntime::avatarVrmSession)
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:consumer>)
endif()
enable_testing()
add_test(NAME installed.vrm_session COMMAND consumer ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
