cmake_minimum_required(VERSION 3.22)
project(InstalledMotionAdapter LANGUAGES CXX)
set(components motion)
if(WITH_VRM)
    list(APPEND components vrm)
endif()
find_package(AvatarRuntime 0.1.0 EXACT CONFIG REQUIRED COMPONENTS ${components})
enable_testing()
add_executable(installedMotionAdapter ../tests.cpp)
target_link_libraries(installedMotionAdapter PRIVATE AvatarRuntime::avatarMotionAdapter)
if(WITH_VRM)
    target_compile_definitions(installedMotionAdapter PRIVATE AR_TEST_VRM)
    target_link_libraries(installedMotionAdapter PRIVATE AvatarRuntime::avatarVrmAdapter)
endif()
if(WIN32)
    add_custom_command(TARGET installedMotionAdapter POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:installedMotionAdapter>)
endif()
add_test(NAME installed.motion_owner_boundary COMMAND installedMotionAdapter)
