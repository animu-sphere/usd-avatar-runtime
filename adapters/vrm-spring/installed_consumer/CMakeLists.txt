cmake_minimum_required(VERSION 3.22)
project(InstalledVrmSpringConsumer LANGUAGES CXX)
find_package(AvatarRuntime 0.1.0 EXACT CONFIG REQUIRED)
if(TARGET vrmRig::vrmRig OR TARGET motionCore::motionCore OR TARGET usd)
    message(FATAL_ERROR "Core-only lookup imported provider dependencies")
endif()
find_package(AvatarRuntime 0.1.0 EXACT CONFIG REQUIRED COMPONENTS vrm_spring)
if(TARGET vrmSchema::vrmSchema OR TARGET AvatarRuntime::avatarVrmAdapter OR TARGET AvatarRuntime::avatarUsdBinding OR TARGET motionSampling::motionSampling)
    message(FATAL_ERROR "Plain spring evaluator imported schema, USD binding, expression or sampling adapters")
endif()
add_executable(consumer ../tests.cpp)
target_link_libraries(consumer PRIVATE AvatarRuntime::avatarVrmSpringAdapter)
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:consumer>)
endif()
enable_testing()
add_test(NAME installed.vrm_spring COMMAND consumer)
