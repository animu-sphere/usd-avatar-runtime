cmake_minimum_required(VERSION 3.22)
project(InstalledVrmSpringUsdConsumer LANGUAGES CXX)
find_package(AvatarRuntime 0.2.0 EXACT CONFIG REQUIRED)
if(TARGET vrmRig::vrmRig OR TARGET vrmSchema::vrmSchema OR TARGET motionUsd::motionUsd)
    message(FATAL_ERROR "Core-only package imported provider dependencies")
endif()
find_package(AvatarRuntime 0.2.0 EXACT CONFIG REQUIRED COMPONENTS vrm_spring_usd)
if(TARGET AvatarRuntime::avatarVrmAdapter OR TARGET AvatarRuntime::avatarMotionAdapter OR TARGET motionSampling::motionSampling)
    message(FATAL_ERROR "Spring stage binding imported expression evaluation or sampling")
endif()
add_executable(consumer ../tests.cpp)
target_link_libraries(consumer PRIVATE AvatarRuntime::avatarVrmSpringUsdBinding)
if(SPRING_ADAPTER_TEST)
    find_package(AvatarRuntime 0.2.0 EXACT CONFIG REQUIRED COMPONENTS vrm_spring)
    target_link_libraries(consumer PRIVATE AvatarRuntime::avatarVrmSpringAdapter)
    target_compile_definitions(consumer PRIVATE AR_TEST_SPRING_ADAPTER)
endif()
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:consumer>)
endif()
enable_testing()
add_test(NAME installed.vrm_spring_usd COMMAND consumer)
