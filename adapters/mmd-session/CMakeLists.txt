find_package(physicsJolt 0.1 CONFIG REQUIRED)
find_package(mmdPhysicsAdapter 0.2 CONFIG REQUIRED)
foreach(api IN ITEMS INITIAL_POSE BODY_PROPERTIES JOINT_SPRINGS)
    if(NOT mmdPhysicsAdapter_${api}_VERSION EQUAL 1)
        message(FATAL_ERROR "MMD session requires installed mmdPhysicsAdapter ${api} API version 1")
    endif()
endforeach()
if(NOT physicsJolt_BACKEND_AVAILABLE)
    message(FATAL_ERROR "MMD session requires an available Jolt backend")
endif()
add_library(avatarMmdSession STATIC session.cpp)
add_library(AvatarRuntime::avatarMmdSession ALIAS avatarMmdSession)
target_compile_features(avatarMmdSession PUBLIC cxx_std_20)
set_target_properties(avatarMmdSession PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(avatarMmdSession PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(avatarMmdSession PUBLIC avatarMmdAdapter avatarUsdBinding
    avatarStageRunnerSession avatarPhysicsUsdResources physicsJolt::physicsJolt)
if(MSVC)
    target_compile_options(avatarMmdSession PRIVATE /W4 /permissive- /utf-8)
endif()
install(TARGETS avatarMmdSession EXPORT AvatarMmdSessionTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT AvatarMmdSessionTargets NAMESPACE AvatarRuntime:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AvatarRuntime)
set(AVATAR_MMD_SESSION_TEST_PMX "" CACHE FILEPATH "Generated morph PMX for direct session tests")
set(AVATAR_MMD_SESSION_TEST_VMD "" CACHE FILEPATH "Generated mixed primary UV/vertex VMD for direct session tests")
if(AVATAR_BUILD_TESTS AND AVATAR_MMD_SESSION_TEST_PMX AND AVATAR_MMD_SESSION_TEST_VMD)
    add_executable(mmdSessionSnapshots tests.cpp)
    target_link_libraries(mmdSessionSnapshots PRIVATE avatarMmdSession)
    add_test(NAME hosts.mmd_session_snapshots COMMAND mmdSessionSnapshots
        "${AVATAR_MMD_SESSION_TEST_PMX}" "${AVATAR_MMD_SESSION_TEST_VMD}")
endif()
