cmake_minimum_required(VERSION 3.22)
project(InstalledMotionUsdConsumer LANGUAGES CXX)
find_package(AvatarRuntime 0.1.0 EXACT CONFIG REQUIRED COMPONENTS motion_usd)
if(TARGET vrmSchema::vrmSchema OR TARGET vrmRig::vrmRig OR TARGET motionSampling::motionSampling)
    message(FATAL_ERROR "Source clip binding must not import format evaluators or sampling")
endif()
add_executable(consumer ../tests.cpp)
target_link_libraries(consumer PRIVATE AvatarRuntime::avatarMotionUsdBinding)
if(WIN32)
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:consumer>)
endif()
enable_testing()
add_test(NAME installed.motion_usd_clip COMMAND consumer)
