cmake_minimum_required(VERSION 3.22)
project(AvatarInstalledConsumer LANGUAGES C)
find_package(AvatarRuntime 0.1.0 EXACT CONFIG REQUIRED)
enable_testing()
add_library(contractProvider SHARED ../contracts/provider.c)
target_link_libraries(contractProvider PRIVATE AvatarRuntime::avatarRuntime)
set_target_properties(contractProvider PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
add_executable(cAbiConsumer ../contracts/c_consumer.c)
target_link_libraries(cAbiConsumer PRIVATE AvatarRuntime::avatarRuntime contractProvider)
set_target_properties(cAbiConsumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(WIN32)
    add_custom_command(TARGET cAbiConsumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E
        copy_if_different $<TARGET_FILE:AvatarRuntime::avatarRuntime> $<TARGET_FILE_DIR:cAbiConsumer>)
endif()
add_test(NAME installed.c_boundary COMMAND cAbiConsumer)
